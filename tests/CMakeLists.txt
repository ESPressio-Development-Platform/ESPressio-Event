cmake_minimum_required(VERSION 3.16)
project(ESPressioEventContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ESPRESSIO_DEPENDENCIES "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Dependency working-branch checkouts")
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
set(EVENT_INCLUDES ../src)
foreach(dependency System Primitive Task Threads Timing Serializable Units Observable)
    list(APPEND EVENT_INCLUDES "${ESPRESSIO_DEPENDENCIES}/ESPressio-${dependency}/src")
endforeach()
# Actual providers, clocks and Thread contracts; only Arduino String declarations
# required by Units are stubbed. The Task harness supplies cooperative host tasks.
list(APPEND EVENT_INCLUDES "${ESPRESSIO_DEPENDENCIES}/ESPressio-Timing/tests/stubs"
    "${ESPRESSIO_DEPENDENCIES}/ESPressio-Task/tests")
foreach(name pool_lifetime type_compile_contract dispatch_lane thread_capability runtime_freeze wire_v1 remote_receipts remote_admission timing_bridge blocking_admission formats_and_contract initialization_rollback remote_contention thread_pause_quiesce)
    add_executable(event_${name} test_event_${name}.cpp)
    target_include_directories(event_${name} PRIVATE ${EVENT_INCLUDES})
    target_compile_options(event_${name} PRIVATE -Wall -Wextra -Werror -fno-rtti -UNDEBUG)
    target_link_libraries(event_${name} PRIVATE Threads::Threads)
    add_test(NAME event_${name} COMMAND event_${name})
    set_tests_properties(event_${name} PROPERTIES TIMEOUT 15)
endforeach()
